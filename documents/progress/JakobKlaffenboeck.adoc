[source, sql]
----
const startButton = document.getElementById('start');
const stopButton = document.getElementById('stop');
const output = document.getElementById('output').querySelector('span');

let recognition;

if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();

    recognition.lang = 'en-US'; // Set language
    recognition.interimResults = true;

    recognition.onresult = (event) => {
        const transcript = Array.from(event.results)
            .map(result => result[0])
            .map(result => result.transcript)
            .join('');
        output.textContent = transcript;
    };

    recognition.onerror = (event) => {
        console.error('Speech Recognition Error:', event.error);
    };
} else {
    alert('Your browser does not support Speech Recognition.');
}

startButton.addEventListener('click', () => {
    if (recognition) recognition.start();
});

stopButton.addEventListener('click', () => {
    if (recognition) recognition.stop();
});

[source, javascript]

function animateChat(positionLength) {
    const queryOutput = document.getElementById("queryOutput");
    const chatHistory = document.getElementById("chatHistory");
    const userInput = document.getElementById("userInput");

    if (positionLength === 0) {
        queryOutput.classList.add("hidden");
        return;
    }

    // Create a new chat entry
    const chatEntry = document.createElement("div");
    chatEntry.classList.add("chatEntry");

    // Add user query
    const userQuery = document.createElement("p");
    userQuery.classList.add("userQuery");
    userQuery.textContent = userInput.value;
    chatEntry.appendChild(userQuery);

    // Add search results
    const resultsContainer = document.createElement("div");
    resultsContainer.classList.add("resultsContainer");
    resultsContainer.innerHTML = queryOutput.innerHTML;
    chatEntry.appendChild(resultsContainer);

    // Append the chat entry to the chat history
    chatHistory.appendChild(chatEntry);

    // Clear the input field
    userInput.value = "";

    queryOutput.classList.add("hidden");
}

Neuer Prompt:

[source, text]
----

ChatGPT – Heating and Plumbing Product Locator

You work at a heating and plumbing wholesale company where customers frequently ask for specific products and their locations within the store. You have access to a document that lists products along with their exact locations.
Your task is to guide customers in finding requested items quickly and accurately.

Core Behavior:

1. Always look up the product name in the provided document.
2. Never invent a location if the product is not found.
- If you don’t find an exact match, simply respond:
> „Ich habe leider keine Informationen über den Stellplatz dieses Produkts.“
3. If the user asks for a product in plural, first **convert it to the singular form** before searching.
- Example:
- “Gestellschrauben” → “Gestellschraube”
- “Dichtungen” → “Dichtung”
- Treat plurals as equivalent to singular.
4. Respond in German using this exact template:
> “Die [Produktname] befindet/befinden sich am Stellplatz [Stellplatz aus Dokument].
> Kann ich Ihnen sonst noch weiterhelfen?”


Examples:

User: “Wo finde ich die Gestellschraube?”
Assistant: “Die Gestellschraube befindet sich am Stellplatz PL-00-00-00.
Kann ich Ihnen sonst noch weiterhelfen?”

User: “Wo finde ich die Gestellschrauben?”
Assistant: “Die Gestellschrauben befinden sich am Stellplatz PL-00-00-00.
Kann ich Ihnen sonst noch weiterhelfen?”


Important:
- Always assume plural product names refer to the same product in singular form.
- Do not guess locations.
- Only answer questions about products listed in the document.
